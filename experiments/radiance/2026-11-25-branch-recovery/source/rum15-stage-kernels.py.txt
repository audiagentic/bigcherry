'''            constexpr uint32_t words = SB == 256u ? 16u : 8u;

            const uint32_t kb = c / SB + (i / words) % kFolds;
            return ((Full || tr * 16u < rows) && kb < nkb8)
                ? ((const uint32_t*)Sa)[((size_t)tr * nkb8 + kb) * words + i % words] : 0u;''')
c('''                    {
                        const uint4 w8 =
                            *(const uint4*)&s_ws[(wave * kFolds + f) * 8u + (lane >> 4) * 4u];
                        const uint32_t d[4] = { w8.x, w8.y, w8.z, w8.w };
#pragma unroll
                        for (uint32_t t = 0; t < 8; ++t)
                            wsr[t] = bscale((uint16_t)(d[t / 2] >> (16 * (t % 2))));
                    }''',
'''                    if constexpr (SB == 256u) {
                        const float* scale = (const float*)&s_ws[

                        for (uint32_t t = 0; t < 8; ++t) wsr[t] = scale[t];
                    } else {
                        const uint4 w8 =
                            *(const uint4*)&s_ws[(wave * kFolds + f) * 8u + (lane >> 4) * 4u];
                        const uint32_t d[4] = { w8.x, w8.y, w8.z, w8.w };
#pragma unroll
                        for (uint32_t t = 0; t < 8; ++t)
                            wsr[t] = bscale((uint16_t)(d[t / 2] >> (16 * (t % 2))));
                    }''')

''','''    const uint8_t* __restrict__ Ap, const void* __restrict__ Sa,                               \\

c('''        /* The lane's eight rows' bf16 scales of this group are consecutive in the tile-major
         * plane: one 16-byte load. */
        const uint4 q4 = *(const uint4*)(Sa + ((size_t)rt * ngrp + g) * 16u + acc_row(lane, 0));
        const uint32_t qd[4] = { q4.x, q4.y, q4.z, q4.w };
#pragma unroll
        for (uint32_t j = 0; j < NT; ++j) {
            const float as = xsr[j][g];
#pragma unroll
            for (uint32_t t = 0; t < 4; ++t) {
                out[j][2 * t] = __builtin_fmaf(acc_val<true>(raw[j][2 * t]),
                                               __uint_as_float(qd[t] << 16) * as, out[j][2 * t]);
                out[j][2 * t + 1] = __builtin_fmaf(acc_val<true>(raw[j][2 * t + 1]),
                                                   __uint_as_float(qd[t] & 0xFFFF0000u) * as,
                                                   out[j][2 * t + 1]);
            }
        }''',
'''        // One F32 scale per weight row at IQ8R's SB=256 group.
        // Ordinary SB=128 I8 keeps the original packed BF16 scale.
        const size_t base = ((size_t)rt * ngrp + g) * 16u + acc_row(lane, 0);
        float ws[8];
        if constexpr (SB == 256u) {
            const float* source = (const float*)Sa + base;
#pragma unroll
            for (uint32_t t = 0; t < 8; ++t) ws[t] = source[t];
        } else {
            const uint16_t* source = (const uint16_t*)Sa + base;
#pragma unroll
            for (uint32_t t = 0; t < 8; ++t) ws[t] = bscale(source[t]);
        }
#pragma unroll
        for (uint32_t j = 0; j < NT; ++j) {
            const float as = xsr[j][g];
#pragma unroll
            for (uint32_t t = 0; t < 8; ++t)
                out[j][t] = __builtin_fmaf(acc_val<true>(raw[j][t]), ws[t] * as, out[j][t]);
        }''')
p.write_text(s)
