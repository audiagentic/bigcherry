"""Isolated experimental IQ8R two-factor U8+BF16 scale layout.

Not a managed patch and not in production. Exact matches required for every
replaced source anchor, so we never drift from the verified FP32 candidate.
"""
from pathlib import Path
d=Path("SCRATCH/rum15-factors-source/libr4d")
def patch(filename,changes):
    p=d/filename
    s=p.read_text()
    for a,b in changes:
        n=s.count(a)
        assert n==1,(filename,n,a[:110])
        s=s.replace(a,b)
    p.write_text(s)
    print("MODIFIED",filename,len(changes),flush=True)

            for (long long r = 0; r < N; ++r) {
                const uint8_t* s1 = r4d_enc::row(pl[0], r);

                const uint8_t* s1 = r4d_enc::row(pl[0], r);
