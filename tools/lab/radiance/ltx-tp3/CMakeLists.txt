cmake_minimum_required(VERSION 3.21)
project(bc_ltx_tp3_reference LANGUAGES CXX)
include(CTest)

if(BUILD_TESTING)
  foreach(name IN ITEMS ltx_tp3_reference_test ltx_tp3_protocol_test ltx_tp3_numerical_test ltx_tp3_linear_test ltx_tp3_rms_test ltx_tp3_handoff_test ltx_tp3_guidance_test ltx_tp3_output_test ltx_tp3_micro_pipeline_test ltx_tp3_i8a8h_test ltx_tp3_builder_shard_test)
    add_executable(${name} ${name}.cpp)
    target_compile_features(${name} PRIVATE cxx_std_20)
    if(name STREQUAL "ltx_tp3_builder_shard_test")
      target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../engines/radiance/patches/rad0060_ltx2_tp3_linear_declare/src)
    endif()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
endif()
