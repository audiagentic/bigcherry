# gfx12 splits the wait counters and names the store one s_wait_storecnt; gfx11 waits on the same counter with
# s_waitcnt_vscnt. Used after a peer store in the all-reduce units, the fused fp8 quantiser and the 2D copy's unit.
asm volatile("s_wait_storecnt 0x0" ::: "memory") ==> asm volatile("s_waitcnt_vscnt null, 0x0" ::: "memory")

# far slower than the tiled row (valid for every M) once a call has more than 16 rows (27B shapes, one XTX, kbench --all-rows:
# M=18 gemv_wide 484 us / decode 1778 us / nt_m64 842 us vs tiled 319 us; M=64 nt_m64 3267 us vs tiled 530 us; M=128 decode 3662 us
# vs tiled 571 us), while the decode row is the fastest up to 16 rows. So: decode keeps M <= 16, gemv_wide and decode_shortk become
# unreachable, and the nt_m64 row drops just below tiled (native/r4d_rows.cpp.rw) so it only serves what tiled cannot.
